min_max_normalize <- function(x) {
  (x - min(x)) / (max(x) - min(x))
}

mtcars_norm<-as.data.frame(lapply(mtcars,min_max_normalize))
head(mtcars_norm)

mtcars$mpg_normalized<-min_max_normalize(mtcars$mpg)
